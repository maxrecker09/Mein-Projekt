\begin{frame}{4.1 b-adische Darstellung reeller Zahlen}
\begin{thm}[b-adische Darstellung reeller Zahlen]
Sei $b \in \N$ mit $b \geq 2$ und $x \in \R \setminus \{0\}$. Dann existieren eindeutig bestimmte Zahlen
\[
E \in \Z,\qquad \sigma \in \{-1,1\},\qquad z_i \in \{0,1,\dots,b-1\}
\]
für $i \in \N \cup \{0\}$ mit $z_0 \neq 0$ und $\{i \in \N \mid z_i \neq b-1\}$ unendlich, so dass
\[
x=\sigma \cdot b^E \cdot \left(\sum_{i=0}^{\infty} z_i b^{-i}\right).
\]
\end{thm}

Diese Darstellung heißt die \textbf{normalisierte b-adische Darstellung} von $x$.
\end{frame}

\begin{frame}{Bestandteile der Darstellung}
\[
x=\sigma \cdot b^E \cdot \left(\sum_{i=0}^{\infty} z_i b^{-i}\right)
\]

\begin{itemize}
    \item $\sigma \in \{-1,1\}$ ist das \textbf{Vorzeichen}.
    \item $b$ ist die \textbf{Basis des Zahlensystems}.
    \item $E \in \Z$ beschreibt die \textbf{Größenordnung} der Zahl.
    \item $z_i$ ist die \textbf{$i$-te Ziffer} mit
    \[
    z_i \in \{0,1,\dots,b-1\}.
    \]
    \item $b^{-i}$ ist der \textbf{Stellenwert} der Ziffer $z_i$.
    \item Also ist
    \[
    z_i \cdot b^{-i}=\text{Ziffer} \cdot \text{Stellenwert}.
    \]
\end{itemize}

\medskip

Beispiele für Basen:
\[
b=10 \text{ (Dezimalsystem)},\quad b=2 \text{ (Binärsystem)},\quad b=16 \text{ (Hexadezimalsystem)}.
\]
\end{frame}

\begin{frame}{Bemerkung zur unendlichen Summe}
Wir verwenden die übliche Notation für Reihen:
\[
\sum_{i=0}^{\infty} a_i := \lim_{n\to\infty}\sum_{i=0}^{n} a_i.
\]

Im vorliegenden Fall betrachten wir
\[
\sum_{i=0}^{\infty} z_i b^{-i}.
\]

\begin{itemize}
    \item Alle Summanden sind nichtnegativ, also sind die Partialsummen monoton wachsend.
    \item Wegen $z_i \leq b-1$ gilt
    \[
    \sum_{i=0}^{\infty} z_i b^{-i}
    \leq
    \sum_{i=0}^{\infty} (b-1)b^{-i}.
    \]
    \item Die rechte Seite ist eine geometrische Reihe:
    \[
    \sum_{i=0}^{\infty} (b-1)b^{-i}
    =
    (b-1)\sum_{i=0}^{\infty} b^{-i}
    =
    (b-1)\cdot \frac{1}{1-\frac1b}
    = b.
    \]
\end{itemize}

Also existiert der Grenzwert und es gilt sogar
\[
1 \leq \sum_{i=0}^{\infty} z_i b^{-i} < b.
\]
\end{frame}

\begin{frame}{Beginn des Beweises: Normierung}
Zunächst ist das Vorzeichen klar:
\[
\sigma = -1 \text{ falls } x<0,
\qquad
\sigma = 1 \text{ falls } x>0.
\]

Außerdem gilt wegen $z_0 \neq 0$:
\[
1 \leq \sum_{i=0}^{\infty} z_i b^{-i} < b.
\]

Daher muss man $x$ zunächst auf den Bereich $[1,b)$ normieren. Dazu wählt man
\[
E := \lfloor \log_b |x| \rfloor
\]
und setzt
\[
a_0 := b^{-E}|x|.
\]

Dann gilt
\[
1 \leq a_0 < b.
\]

\medskip

Idee: Wir schreiben also
\[
x=\sigma \cdot b^E \cdot a_0
\]
mit einer normierten Mantisse $a_0 \in [1,b)$.
\end{frame}

\begin{frame}{Rekursive Konstruktion der Ziffern}
Nun werden die Ziffern schrittweise aus $a_0$ gewonnen.

Für $i \in \N \cup \{0\}$ definiert man rekursiv:
\[
z_i := \lfloor a_i \rfloor
\qquad\text{und}\qquad
a_{i+1} := b(a_i-z_i).
\]

\begin{itemize}
    \item $z_i$ ist der ganzzahlige Teil von $a_i$,
    also die nächste Ziffer.
    \item $a_i-z_i$ ist der Nachkommateil.
    \item Durch Multiplikation mit $b$ wird die nächste Stelle nach vorne geholt.
\end{itemize}

\medskip

Beispiel in Basis $10$ mit $a_0=3{,}14$:
\[
z_0=\lfloor 3{,}14\rfloor =3,
\qquad
a_1=10(3{,}14-3)=1{,}4,
\]
\[
z_1=\lfloor 1{,}4\rfloor =1,
\qquad
a_2=10(1{,}4-1)=4.
\]

So entstehen Schritt für Schritt die Ziffern der Darstellung.
\end{frame}

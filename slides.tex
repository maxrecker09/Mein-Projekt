\begin{frame}{Satz 4.1: b-adische Darstellung reeller Zahlen}

\begin{thm}
Sei \( b \in \N \) mit \( b \geq 2 \). Sei \( x \in \R \setminus \{0\} \). Dann existieren eindeutig bestimmte Zahlen

\[
E \in \Z,\quad \sigma \in \{-1,1\},\quad z_i \in \{0,1,\dots,b-1\}
\]

für \( i \in \N \cup \{0\} \) mit

\[
\{\, i \in \N \mid z_i \neq b-1 \,\} \text{ unendlich}, \quad z_0 \neq 0
\]

und

\[
x=\sigma \cdot b^E \cdot \left(\sum_{i=0}^{\infty} z_i \, b^{-i}\right).
\]

Dies heißt die (normalisierte) b-adische Darstellung von \(x\).
\end{thm}

\end{frame}

%------------------------------------------------
\begin{frame}{Bestandteile von \(x\)}

\[
x=\sigma \cdot b^E \cdot \left(\sum_{i=0}^{\infty} z_i\, b^{-i}\right)
\]

\vspace{0.4cm}

\begin{itemize}
    \item \(x\): die Zahl, die dargestellt werden soll
    
    \item \(\sigma \in \{-1,1\}\): Vorzeichen
    
    \item \(b\): Basis des Zahlensystems
    
    \item \(E \in \Z\): Exponent / Größenordnung
    
    \item \(z_i\): \(i\)-te Ziffer der Zahl
    
    \item \(b^{-i}\): Stellenwert der jeweiligen Ziffer
\end{itemize}

\vspace{0.4cm}

\textbf{Beispiel:}

\[
3{,}1415
\quad\Rightarrow\quad
z_0=3,\; z_1=1,\; z_2=4,\; z_3=1,\; z_4=5
\]

\vspace{0.3cm}

\[
\Rightarrow\quad
\text{Zahl = Vorzeichen \(\cdot\) Größe \(\cdot\) Mantisse}
\]

\end{frame}

%------------------------------------------------
\begin{frame}{Behauptung und Beginn des Beweises}

\scriptsize

\textbf{Behauptung:} Es gilt

\[
1 \leq \sum_{i=0}^{\infty} z_i b^{-i}
<
\sum_{i=0}^{\infty}(b-1)b^{-i}
= b
\]

d.h. der Grenzwert existiert, weil die Folge monoton wachsend ist und durch \(b\) nach oben beschränkt.

\vspace{0.35cm}

\textbf{Bew. der Aussage:}

\begin{enumerate}
\item Sei \(x\in\R\setminus\{0\}\).

Dann gilt offensichtlich:

\[
\sigma=
\begin{cases}
-1,& x<0\\[0.2em]
1,& x>0
\end{cases}
\]

Für

\[
z_i\in\{0,1,\dots,b-1\},
\qquad i\in\N\cup\{0\}
\]

mit

\[
\{\,i\in\N \mid z_i\neq b-1\,\}\neq\emptyset
\qquad\text{und}\qquad
z_0\neq 0
\]

gilt.

\end{enumerate}

\end{frame}

%------------------------------------------------
\begin{frame}{Warum gilt \(1 \leq \sum z_i b^{-i}\)?}

\scriptsize

\textbf{(2) Warum ist}

\[
1 \leq \sum_{i=0}^{\infty} z_i b^{-i}\; ?
\]

\vspace{0.3cm}

Da \(z_0\neq 0\) und \(z_0\in\{0,1,\dots,b-1\}\), folgt:

\[
z_0\geq 1.
\]

\vspace{0.2cm}

Damit gilt:

\[
\sum_{i=0}^{\infty} z_i b^{-i}
=
z_0+\sum_{i=1}^{\infty} z_i b^{-i}
\geq z_0\geq 1.
\]

\vspace{0.3cm}

\textbf{Gleichheit genau dann, wenn}

\[
\forall i\in\N:\ z_i=0.
\]

Dann:

\[
\sum_{i=0}^{\infty} z_i b^{-i}=1.
\]

\vspace{0.2cm}

Sonst immer größer als \(1\).

\end{frame}

%------------------------------------------------
\begin{frame}{Warum ist \(\sum (b-1)b^{-i}=b\)?}

\scriptsize

\textbf{(3) Warum ist}

\[
\sum_{i=0}^{\infty}(b-1)b^{-i}=b\; ?
\]

\vspace{0.4cm}

\[
\sum_{i=0}^{\infty}(b-1)b^{-i}
=
(b-1)\sum_{i=0}^{\infty} b^{-i}
\]

\vspace{0.3cm}

Dies ist eine geometrische Reihe mit Quotient

\[
q=\frac{1}{b}.
\]

Daher:

\[
\sum_{i=0}^{\infty} b^{-i}
=
\frac{1}{1-\frac{1}{b}}
=
\frac{b}{b-1}.
\]

\vspace{0.3cm}

Einsetzen ergibt:

\[
(b-1)\cdot \frac{b}{b-1}=b.
\]

\end{frame}

%------------------------------------------------
\begin{frame}{Ziel der Normierung}

\textbf{Behauptung:}
\[
E := \left\lfloor \log_b |x| \right\rfloor
\]

\vspace{0.4cm}

Wir wollen \(x\) so schreiben, dass

\[
x=\sigma \cdot b^E \cdot a_0
\qquad \text{mit} \qquad 1 \leq a_0 < b.
\]

Dann folgt:

\[
a_0=b^{-E}|x|.
\]

\vspace{0.4cm}

Gesucht ist also ein Exponent \(E\), für den gilt:

\[
1 \leq b^{-E}|x| < b.
\]

Umformen liefert:

\[
1 \leq \frac{|x|}{b^E} < b
\quad \Longleftrightarrow \quad
b^E \leq |x| < b^{E+1}.
\]

\vspace{0.4cm}

Das bedeutet:

\begin{itemize}
    \item Finde eine ganze Zahl \(E\), sodass \(|x|\)
    zwischen zwei benachbarten Potenzen von \(b\) liegt.
\end{itemize}

\end{frame}

%------------------------------------------------
\begin{frame}{Bestimmung des Exponenten \(E\)}

Aus

\[
b^E \leq |x| < b^{E+1}
\]

wenden wir den Logarithmus zur Basis \(b\) an:

\[
\log_b(b^E) \leq \log_b(|x|) < \log_b(b^{E+1}).
\]

Mit

\[
\log_b(b^E)=E,
\qquad
\log_b(b^{E+1})=E+1
\]

folgt:

\[
E \leq \log_b(|x|) < E+1.
\]

\vspace{0.4cm}

Damit muss gelten:

\[
E=\left\lfloor \log_b(|x|)\right\rfloor.
\]

\vspace{0.5cm}

\textbf{Grund:}

Für

\[
a_0=b^{-E}|x|=\frac{|x|}{b^E}
\]

gilt dann automatisch:

\[
1 \leq a_0 < b.
\]

Also liefert
\[
E=\left\lfloor \log_b(|x|)\right\rfloor
\]
genau den passenden Exponenten für die normierte Darstellung.

\end{frame}

\begin{frame}{Darstellung von \(a_0\) mit Restglied}

\scriptsize

\textbf{Behauptung:} Für alle \(n\in\N\cup\{0\}\) gilt

\[
a_0=
\sum_{i=0}^{n} z_i b^{-i}
+
a_{n+1}b^{-n-1}.
\]

\vspace{0.2cm}

Dabei ist:

\begin{itemize}
\item \(\sum_{i=0}^{n} z_i b^{-i}\): bekannte Ziffern / Partialsumme
\item \(a_{n+1}b^{-n-1}\): Restglied
\end{itemize}

\vspace{0.2cm}

Der verbleibende Rest wird ab dem Index \(n+1\) zusammengefasst.

\vspace{0.3cm}

\textbf{Ziel:}

\[
a_0=\sum_{i=0}^{\infty} z_i b^{-i}.
\]

Das Restglied verschwindet für \(n\to\infty\).

\vspace{0.3cm}

\textbf{Wichtig:}

Falls

\[
1\leq x<b,
\]

dann gilt bereits

\[
a_0=x,
\]

also ist keine zusätzliche Normierung nötig.

\end{frame}

\begin{frame}{Beweis per Induktion: Induktionsanfang}

\small

\textbf{Induktionsanfang: \(n=0\)}

Zu zeigen:

\[
a_0=z_0 b^{0}+a_1 b^{-1}.
\]

\[
a_0=z_0+\frac{a_1}{b}.
\]

Wir wissen:

\[
a_1=b(a_0-z_0).
\]

Einsetzen liefert:

\[
a_0=
z_0+\frac{b(a_0-z_0)}{b}.
\]

\[
a_0=
z_0+(a_0-z_0)=a_0.
\]

\hfill \(\checkmark\)

\end{frame}

%------------------------------------------------
\begin{frame}{Beweis per Induktion: Induktionsvoraussetzung}

\small

\textbf{Induktionsvoraussetzung \((n-1\rightarrow n)\):}

Zu zeigen:

\[
a_0=
\sum_{i=0}^{n} z_i b^{-i}
+
a_{n+1} b^{-n-1}.
\]

\vspace{0.5cm}

Also lautet die Induktionsannahme:

\[
a_0=
\sum_{i=0}^{n-1} z_i b^{-i}
+
a_n b^{-n}.
\]

\end{frame}

%------------------------------------------------
%------------------------------------------------
\begin{frame}{I.S. (Rekursionsformel)}

\scriptsize

Aus

\[
a_{n+1}=b(a_n-z_n)
\]

folgt

\[
a_n=z_n+\frac{a_{n+1}}{b}.
\]

\vspace{0.3cm}

Dann:

\[
a_0=
\sum_{i=0}^{n-1} z_i b^{-i}
+
\left(z_n+\frac{a_{n+1}}{b}\right)b^{-n}
\]

\vspace{0.3cm}

\[
\Longleftrightarrow\quad
a_0=
\sum_{i=0}^{n-1} z_i b^{-i}
+
z_n b^{-n}
+
a_{n+1}b^{-n-1}
\]

\vspace{0.3cm}

\[
\Longleftrightarrow\quad
a_0=
\sum_{i=0}^{n} z_i b^{-i}
+
a_{n+1}b^{-n-1}
\]

\hfill \(\checkmark\)

\end{frame}

\begin{frame}{Übergang zum Grenzwert}

\small

Aus

\[
x=\sigma \cdot b^E \cdot a_0
\]

und

\[
a_0=
\sum_{i=0}^{n} z_i b^{-i}
+
a_{n+1}b^{-n-1}
\]

für jedes \(n\in\N\cup\{0\}\) folgt:

\[
x=
\sigma \cdot b^E \cdot
\lim_{n\to\infty}
\left(
\sum_{i=0}^{n} z_i b^{-i}
\right),
\]

sofern das Restglied verschwindet.

\vspace{0.3cm}

Also müssen wir zeigen:

\[
a_{n+1}b^{-n-1}\to 0
\qquad (n\to\infty).
\]

\vspace{0.3cm}

Per Induktion gilt die Darstellung bereits für jedes endliche \(n\).  
Nun geht man zum Grenzwert über.

\end{frame}

%------------------------------------------------
\begin{frame}{Verschwinden des Restglieds}

\small

Zu zeigen:

\[
a_{n+1}b^{-n-1}\to 0.
\]

Dann bleibt:

\[
a_0=
\lim_{n\to\infty}\sum_{i=0}^{n} z_i b^{-i}
=
\sum_{i=0}^{\infty} z_i b^{-i}.
\]

\vspace{0.4cm}

Es wurde gezeigt:

\[
1\leq a_{n+1}<b.
\]

Daraus folgt:

\[
0\leq a_{n+1}b^{-n-1}
<
b\cdot b^{-n-1}
=
b^{-n}.
\]

Da \(b^{-n}\to 0\), gilt mit dem Sandwichprinzip:

\[
a_{n+1}b^{-n-1}\to 0.
\]

\end{frame}

\begin{frame}{Eindeutigkeit der normalisierten Darstellung}

\small

\textbf{Annahme:}

\[
\{\, i\in\N \mid z_i\neq b-1 \,\}
\]

sei endlich.

Dann existiert ein \(n_0\in\N\) mit

\[
z_i=b-1
\qquad \text{für alle } i>n_0.
\]

\vspace{0.4cm}

\textbf{Behauptung:}

Für jedes \(x\in\R\setminus\{0\}\) gibt es genau eine normalisierte Darstellung

\[
x=\sigma \cdot b^E \cdot \sum_{i=0}^{\infty} z_i b^{-i},
\]

mit

\[
\sigma\in\{-1,1\},\quad E\in\Z,\quad z_0\neq 0.
\]

Die Bedingung verhindert doppelte Darstellungen mit endlosem \(b-1\)-Schwanz.

\end{frame}

%------------------------------------------------
\begin{frame}{1. Existenz der Darstellung}

\textbf{Existenz:}

Zu jeder Zahl \(x\) lassen sich passende Ziffern \(z_i\) wählen, sodass

\[
x=\sigma \cdot b^E \cdot \sum_{i=0}^{\infty} z_i b^{-i}
\]

gilt.

\vspace{0.5cm}

\textbf{Idee des Beweises:}

\begin{itemize}
    \item Die Zahl wird schrittweise in Basis \(b\) zerlegt.
    
    \item Für jede Stelle wird eine passende Ziffer \(z_i\) bestimmt.
    
    \item Dadurch entsteht die unendliche Reihe
    
    \[
    \sum_{i=0}^{\infty} z_i b^{-i}.
    \]
    
    \item Würde ab einem gewissen Punkt nur noch \(b-1\) auftreten, entstünde eine alternative Darstellung.
    
    \item Deshalb wird dieser Fall ausgeschlossen.
\end{itemize}

\vspace{0.5cm}

\[
\Rightarrow \text{Also existiert eine normalisierte Darstellung.}
\]

\end{frame}

%------------------------------------------------
\begin{frame}{2. Eindeutigkeit der Darstellung}

\textbf{Beweisidee zur Eindeutigkeit:}

Angenommen, es gäbe zwei verschiedene normalisierte Darstellungen derselben Zahl.

\vspace{0.4cm}

Dann sei \(n\) die erste Stelle, an der sich die Ziffern unterscheiden.

\begin{itemize}
    \item Vor der Stelle \(n\) sind beide Darstellungen identisch.
    
    \item An der Stelle \(n\) ist eine Darstellung größer als die andere.
    
    \item Dieser Unterschied besitzt den Stellenwert \(b^{-n}\).
    
    \item Die späteren Stellen haben nur kleinere Gewichte
    \(\left(b^{-n-1}, b^{-n-2}, \dots \right)\)
    und können den Unterschied nicht mehr vollständig ausgleichen.
\end{itemize}

\vspace{0.5cm}

Daher können beide Summen nicht gleich sein.

\[
\Rightarrow \text{Die Darstellung ist eindeutig.}
\]

\end{frame}

%------------------------------------------------
\begin{frame}{Merksatz und Beispiele}

\textbf{Merksatz:}

Wenn an der ersten unterschiedlichen Stelle eine Ziffer kleiner ist, dann kann der Rest diesen Unterschied nie vollständig aufholen.

\vspace{0.6cm}

\textbf{Beispiele für \(b=2\):}

\[
\frac{1}{3}
=
2^{-2}\left(2^0+2^{-2}+2^{-4}+2^{-6}+\dots\right)
=
(0,\overline{01})_2
\]

\vspace{0.6cm}

\[
8{,}1
=
2^3\left(2^0+2^{-7}+2^{-8}+2^{-11}+2^{-12}+\dots\right)
=
(1000,0\overline{0011})_2
\]

\end{frame}

%------------------------------------------------
\begin{frame}{4.2 Maschinenzahlen}

\large

\textbf{Motivation / Zusammenhang:}

\vspace{0.5cm}

Da in einem Rechner nur endlich viele Stellen einer reellen Zahl dargestellt werden können,

nutzt man in Anlehnung an Satz 4.1  
(normalisierte \(b\)-adische Darstellung von \(x\))

die folgende Darstellung.

\end{frame}

%------------------------------------------------
\begin{frame}{Definition 4.2.1: Maschinenzahl}

\tiny

Seien \(b,m\in\N\) mit \(b\geq 2\).

Eine \(b\)-adische \(m\)-stellige normalisierte Gleitkommazahl hat die Form

\[
x=\sigma \cdot b^E \cdot
\left(\sum_{i=0}^{m-1} z_i b^{-i}\right)
\]

mit

\[
\sigma\in\{-1,1\},
\qquad
E\in\Z,
\]

\[
z_0\neq 0,
\qquad
z_i\in\{0,\dots,b-1\}
\quad \text{für alle } i\in\N\cup\{0\}.
\]

\vspace{0.4cm}

\(\Rightarrow\) Der einzige Unterschied zu Satz 4.1 ist:

Die Reihe läuft jetzt nur noch bis \(m-1\), da wir eine \(m\)-stellige Zahl haben.

\vspace{0.4cm}

\textbf{Warum nicht bis \(m\)?}

Weil wir bei \(0\) anfangen:

\[
i=0 \text{ (1. Stelle), }\;
i=1 \text{ (2. Stelle), }\;
\dots,\;
i=m-1 \text{ (m-te Stelle).}
\]

\end{frame}

%------------------------------------------------
\begin{frame}{Beispiel}

\large

\textbf{1)} Für \(b=10\) hat die Zahl \(136{,}5\) die normalisierte 5-stellige Darstellung:

\[
136{,}5 = 1{,}3650 \cdot 10^2
\]

\vspace{0.5cm}

In C++-Notation schreibt sich das als:

\[
1.365\texttt{e}+2
\]

\end{frame}

%------------------------------------------------
\begin{frame}{Maschinenzahlbereich}

\small

Für die Darstellung im Rechner sind geeignete Werte für \(b, m\) und den zulässigen Bereich

\[
\{E_{\min}, \dots, E_{\max}\}
\]

für \(E\) zu wählen.

\vspace{0.3cm}

Dies bestimmt den Maschinenzahlbereich:

\[
F(b,m,E_{\min},E_{\max})
\]

\vspace{0.4cm}

\textbf{Bedeutung:}

\begin{itemize}
\item \(b\): Basis des Zahlensystems

\item \(m\): Anzahl der Mantissenstellen

\item \(E_{\min}, E_{\max}\): erlaubter Exponentenbereich
\end{itemize}

\vspace{0.4cm}

In dieser Menge liegen Zahlen der Form:

\[
x=\sigma \cdot b^E \cdot \left(\sum_{i=0}^{m-1} z_i b^{-i}\right),
\]

mit erlaubten Exponenten, genau \(m\) Stellen und normalisiert, sowie zusätzlich \(0\).

\vspace{0.4cm}

\textbf{Beispiel:}

\[
-3 \leq E \leq 4
\]

\end{frame}

%------------------------------------------------
\begin{frame}{Definition des Maschinenzahlbereichs}

\small

Der Maschinenzahlbereich ist die Menge der \(b\)-adischen, \(m\)-stelligen,
normalisierten Gleitkommazahlen mit Exponent

\[
E\in\{E_{\min},\dots,E_{\max}\},
\]

zuzüglich der Zahl \(0\).

\vspace{0.5cm}

\textbf{Hinweis:}

Die \(0\) muss extra hinzugefügt werden, da sie wegen der Normalisierung
(\(z_0\neq 0\)) nicht automatisch in der Menge enthalten ist.

\vspace{0.8cm}

Die Elemente eines (festen) Maschinenzahlbereichs heißen
\textbf{Maschinenzahlen}.

\end{frame}

%------------------------------------------------
\begin{frame}{Beispiel für Maschinenzahlen}

\large

\textbf{Beispiel:}

\[
b=10,\quad m=3,\quad E\in\{-1,0,1\}
\]

\vspace{0.5cm}

Dann sind mögliche Maschinenzahlen z.\,B.:

\[
1{,}23\cdot 10^1 = 12{,}3
\]

\[
9{,}99\cdot 10^{-1} = 0{,}999
\]

\[
-4{,}56\cdot 10^0 = -4{,}56
\]

\vspace{0.5cm}

Und immer auch:

\[
0
\]

\end{frame}
%------------------------------------------------
\begin{frame}{Anwendung in der Praxis}

\small

Im 1985 verabschiedeten IEEE-Standard 754 wird beispielsweise
ein Datentyp \texttt{double} definiert mit

\[
b=2,\quad m=53,\quad E\in\{-1022,\dots,1023\}.
\]

\vspace{0.4cm}

Dabei steht IEEE für:

\textit{The Institute of Electrical and Electronics Engineers}

\vspace{0.5cm}

Wir schreiben:

\[
F_{\text{double}}=F(2,53,-1022,1023).
\]

\vspace{0.5cm}

Zur Darstellung dieser Zahlen reichen 64 Bits aus:

\begin{itemize}
\item 1 Bit für das Vorzeichen
\item 52 Bits für die Mantisse
\item 11 Bits für den Exponenten
\end{itemize}

\end{frame}

%------------------------------------------------
\begin{frame}{IEEE 754: Hidden Bit}

\small

Wir schreiben:

\[
F_{\text{double}}=F(2,53,-1022,1023).
\]

\vspace{0.4cm}

Zur Darstellung dieser Zahlen reichen 64 Bits aus:

\begin{itemize}
\item 1 Bit für das Vorzeichen
\item 52 Bits \(z_1,\dots,z_{52}\) für die Mantisse
\item 11 Bits für den Exponenten
\end{itemize}

\vspace{0.6cm}

Da wegen der Normalisierung das nullte Bit \(z_0\) immer den Wert \(1\) hat,
kann dieses weggelassen werden.

\vspace{0.3cm}

Es wird daher als \textbf{hidden bit} bezeichnet.

\vspace{0.4cm}

Deshalb genügen 52 Mantissenbits bei \(m=53\).

\end{frame}

%------------------------------------------------
\begin{frame}{IEEE 754: Bias-Darstellung}

\small

Für den Exponenten wird die sogenannte \textbf{Bias-Darstellung} benutzt:

Eine Konstante (Bias) wird hinzuaddiert, um alle Exponenten positiv zu speichern.

\vspace{0.6cm}

Im konkreten Fall \texttt{double} wird

\[
1023
\]

hinzuaddiert.

\vspace{0.5cm}

Beispiele:

\[
E=0 \;\rightarrow\; 1023
\]

\[
E=5 \;\rightarrow\; 1028
\]

\[
E=-3 \;\rightarrow\; 1020
\]

\vspace{0.6cm}

Für den C++-Datentyp \texttt{double} schreibt der Standard nicht exakt vor,
wie dies intern umgesetzt wird.

Üblicherweise erfolgt die Implementierung jedoch gemäß IEEE 754.

Davon gehen wir im Folgenden aus.

\end{frame}

%------------------------------------------------
\begin{frame}{Bit-Aufbau der Gleitkommadarstellung}

\small

Die Abspeicherung in Bits für die normalisierte Gleitkommadarstellung sieht wie folgt aus:

\vspace{0.7cm}

\[
\begin{array}{|c|c|c|}
\hline
\textbf{1 Bit} & \textbf{11 Bits} & \textbf{52 Bits} \\
\hline
\dfrac{1+\sigma}{2} & E+1023 & z_1,z_2,\dots,z_{52} \\
\hline
\end{array}
\]

\vspace{0.8cm}

\textbf{Bedeutung:}

\begin{itemize}
\item Vorzeichenbit:
\[
\sigma=-1 \Rightarrow \frac{1+\sigma}{2}=0,
\qquad
\sigma=1 \Rightarrow \frac{1+\sigma}{2}=1
\]

\item Exponent wird mit Bias gespeichert: \(E+1023\)

\item Mantisse speichert die Bits \(z_1,\dots,z_{52}\)
\end{itemize}

\end{frame}

\begin{frame}{Spezielle Exponentenwerte}

\small

Die zwei freien Exponenten \(-1023\) und \(1024\)

(bzw. \(0\) und \(2047\) in Bias-Darstellung)

werden für Sonderfälle benutzt.

\vspace{0.5cm}

\textbf{Verwendung:}

\begin{itemize}
\item \(\pm 0\)

\item \(\pm \infty\)

\item subnormale Zahlen  
(Exponent kleiner als \(-1022\))

\item NaN (\textit{Not a Number})
\end{itemize}

\vspace{0.5cm}

NaN entsteht z.\,B. bei unzulässigen Rechenoperationen wie

\[
\log(-1).
\]

\vspace{0.4cm}

Ebenso können Werte größer als \(1023\) über spezielle Kodierungen dargestellt werden.

\end{frame}

\begin{frame}{Range eines Maschinenzahlbereichs}

\small

Wir bezeichnen mit

\[
\operatorname{range}(F):=
\left[
\min\{\,f\in F \mid f>0\,\},
\max F
\right]
\]

das kleinste Intervall, in dem alle darstellbaren positiven Zahlen liegen.

\end{frame}


%------------------------------------------------
\begin{frame}{4.3 Rundungen}

\small

Da es nur endlich viele Maschinenzahlen gibt,
muss man sich mit gerundeten Darstellungen
von reellen Zahlen zufrieden geben.

\end{frame}

%------------------------------------------------
\begin{frame}{Definition 4.5: Rundung}

\small

Sei

\[
F=F(b,m,E_{\min},E_{\max})
\]

ein Maschinenzahlbereich.

\vspace{0.4cm}

Eine Abbildung

\[
\operatorname{rd}:\mathbb{R}\to F
\]

heißt \textbf{Rundung} (zu \(F\)), wenn für alle \(x\in\mathbb{R}\) gilt:

\[
|x-\operatorname{rd}(x)|
=
\min_{a\in F}|x-a|.
\]

\vspace{0.6cm}

\textbf{Bedeutung:}

\begin{itemize}
\item \(x\): reelle Zahl
\item \(\operatorname{rd}(x)\): gerundete Maschinenzahl
\item gewählt wird die Zahl aus \(F\), die am nächsten an \(x\) liegt
\end{itemize}

\end{frame}

%------------------------------------------------
\begin{frame}{Mehrere mögliche Rundungen}

\small

Man beachte, dass es mehrere Rundungen gibt.

\vspace{0.2cm}

Zum Beispiel kann in

\[
F(10,2,0,2)
\]

die Zahl

\[
12{,}5
\]

auf \(12\) oder \(13\) gerundet werden.

\vspace{0.5cm}

\textbf{Sonderfall:} \(x\) liegt genau zwischen zwei Zahlen.

\vspace{0.4cm}

Bei kaufmännischer Rechnung wird \(x\) im Zweifel zur betragsmäßig größeren Zahl gerundet.

\end{frame}

%------------------------------------------------
\begin{frame}{IEEE 754: Runden zur geraden Zahl}

\small

IEEE 754 fordert:

\begin{center}
$x$ wird so gerundet, dass die letzte Stelle gerade wird.
\end{center}

\vspace{0.2cm}

Also zum Beispiel:

\[
\operatorname{rd}(12{,}5)=12
\qquad\text{und}\qquad
\operatorname{rd}(13{,}5)=14
\]

\vspace{0.5cm}

Ein Grund dafür ist, dass bei Rechnungen wie

\[
(\dots(((12+0{,}5)-0{,}5)+0{,}5)-\dots)
\]

sonst abwechselnd auf- und abgerundet wird.

\vspace{0.2cm}

Bei kaufmännischer Rundung divergiert diese Reihe.

\vspace{0.5cm}

\textbf{Vorteil der Regel:}

Die Regel ,,zur geraden Zahl runden`` wurde gewählt, weil sich Rundungsfehler statistisch besser ausgleichen.

\end{frame}

%------------------------------------------------
\begin{frame}{Definition 4.6: Absoluter und relativer Fehler}

\small

\textbf{Definition 4.6.}
Es seien \(x,\tilde{x}\in\mathbb{R}\), wobei \(\tilde{x}\) eine Näherung von \(x\) sei.

\vspace{0.5cm}

Dann heißt

\[
|x-\tilde{x}|
\]

der \textbf{absolute Fehler}.

\vspace{0.6cm}

Falls \(x\neq 0\), heißt

\[
\left|\frac{x-\tilde{x}}{x}\right|
\]

der \textbf{relative Fehler}.

\vspace{0.5cm}

\begin{itemize}
\item Absoluter Fehler = tatsächliche Differenz.
\item Relativer Fehler = proportionale Abweichung bezogen auf \(x\).
\end{itemize}

\end{frame}

%------------------------------------------------
\begin{frame}{Beispiel: Absoluter und relativer Fehler}

\small

\textbf{Beispiel:}

\[
x=3{,}145
\qquad\text{und}\qquad
\tilde{x}=3{,}14
\]

\vspace{0.5cm}

\textbf{Absoluter Fehler:}

\[
|x-\tilde{x}|
=
|3{,}145-3{,}14|
=
0{,}005
\]

\vspace{0.5cm}

\textbf{Relativer Fehler:}

\[
\left|\frac{x-\tilde{x}}{x}\right|
=
\frac{0{,}005}{3{,}145}
\approx
0{,}0016
\]

\vspace{0.5cm}

\textbf{Als Prozentwert:}

\[
100\cdot 0{,}0016 = 0{,}16\%
\]

Das ist die prozentuale Abweichung von \(\tilde{x}\) zu \(x\).

\end{frame}

%------------------------------------------------
\begin{frame}{Maschinengenauigkeit}

\small

Die \textbf{Maschinengenauigkeit} ist der maximale relative Fehler,  
der bei einer Rundung von Zahlen im darstellbaren Bereich auftreten kann.

\vspace{0.6cm}

Also:

\begin{center}
größte relative Abweichung, die durch Rundung entstehen kann
\end{center}

\vspace{0.6cm}

Bedeutung:

\begin{center}
schlimmste prozentuale Ungenauigkeit,  
die dein Zahlensystem noch verursacht
\end{center}

\end{frame}

%------------------------------------------------
\begin{frame}{eps-Definition}

\scriptsize

\textbf{Definition:} Sei \(F\) ein Maschinenzahlbereich.  
Die Maschinengenauigkeit von \(F\) wird mit \(\operatorname{eps}(F)\) bezeichnet und ist definiert durch

\vspace{0.4cm}

\[
\operatorname{eps}(F):=
\sup \left\{
\left|\frac{x-\operatorname{rd}(x)}{x}\right|
\;\middle|\;
x\in\mathbb{R},\;
|x|\in \operatorname{range}(F),\;
\operatorname{rd}\text{ Rundung zu }F
\right\}.
\]

\vspace{0.5cm}

\small

\begin{itemize}
\item \(\sup\) = kleinste obere Schranke / größter möglicher Grenzwert.
\item Das ist der größte relative Fehler, der auftreten kann.
\item Genau dieser Wert beschreibt die Maschinengenauigkeit.
\end{itemize}

\end{frame}
%------------------------------------------------
\begin{frame}{Bedeutung von \(\operatorname{range}(F)\)}

\small

\[
\operatorname{range}(F)
=
\{x\in\mathbb{R}\;|\;x \text{ liegt im darstellbaren Grenzbereich}\}
\]

\vspace{0.6cm}

\textbf{Anschaulich bedeutet das:}

\begin{itemize}
\item darstellbarer Zahlenbereich als reelle Werte
\item alle Zahlen zwischen der kleinsten und größten darstellbaren Zahl
\item nur in diesem Bereich wird der Rundungsfehler betrachtet
\end{itemize}

\vspace{0.5cm}

\textbf{Beispiel:}

Wenn ein System nur Zahlen von \(10^{-5}\) bis \(10^{5}\) speichern kann, dann liegt genau dieses Intervall in \(\operatorname{range}(F)\).

\end{frame}

%------------------------------------------------
\begin{frame}{Satz 4.8}

\small

\textbf{Satz 4.8}

Für jeden Maschinenzahlbereich

\[
F = F(b,m,E_{\min},E_{\max})
\qquad\text{mit}\qquad
E_{\min}<E_{\max}
\]

gilt:

\vspace{0.6cm}

\[
\operatorname{eps}(F)=\frac{1}{1+2b^{\,m-1}}
\]

\vspace{0.6cm}

\textbf{Das hängt ab von:}

\begin{itemize}
\item Basis \(b\)
\item Mantissenlänge \(m\)
\end{itemize}

\vspace{0.3cm}

Je größer \(m\), desto genauer das System.

\end{frame}
%------------------------------------------------
\begin{frame}{Beweis zu Satz 4.8}

\scriptsize

\textbf{Beweis:}

Für

\[
x=b^{E_{\min}}\left(1+\frac12\,b^{-m+1}\right)
\]

ist \(x\in \operatorname{range}(F)\) und für jede Rundung \(\operatorname{rd}\) zu \(F\) gilt:

\vspace{0.4cm}

\[
\left|\frac{x-\operatorname{rd}(x)}{x}\right|
=
\frac{|x-\operatorname{rd}(x)|}{x}
\]

\vspace{0.3cm}

\[
=
\frac{b^{E_{\min}}\cdot \frac12\, b^{-m+1}}
{b^{E_{\min}}\left(1+\frac12\,b^{-m+1}\right)}
\]

\vspace{0.3cm}

\[
=
\frac{\frac12\,b^{-m+1}}
{1+\frac12\,b^{-m+1}}
\]

\vspace{0.3cm}

\[
=
\frac{1}{1+2b^{m-1}}
\]

\vspace{0.5cm}

\textbf{Idee:}

\begin{itemize}
\item Genau in der Mitte ist der absolute Fehler maximal.
\item Bei kleinem \(x\) wird der relative Fehler am größten.
\end{itemize}

\end{frame}

%------------------------------------------------
\begin{frame}{Folgerung zu \(\operatorname{eps}(F)\)}

\small

Woraus

\[
\operatorname{eps}(F)\;\geq\;\frac{1}{1+2b^{\,m-1}}
\]

folgt.

\vspace{0.6cm}

\textbf{Interpretation:}

\begin{itemize}
\item \(\operatorname{eps}(F)\) ist der größte relative Fehler, der auftreten kann.
\item Wir haben bereits einen konkreten Fehler gefunden.
\item Deshalb muss \(\operatorname{eps}(F)\) mindestens so groß sein.
\end{itemize}

\vspace{0.5cm}

\textbf{Anschaulich:}

\begin{center}
Der schlimmste mögliche Rundungsfehler ist mindestens  
so groß wie der gefundene Fehler.
\end{center}

\end{frame}
%------------------------------------------------
\begin{frame}{Beweis (andere Richtung)}

\small

Für die andere Richtung genügt es, positive Zahlen \(x\) zu betrachten.

\vspace{0.3cm}

\textit{Warum?}  
Negative Zahlen haben denselben relativen Fehler.

\vspace{0.5cm}

Sei also

\[
x \in \operatorname{range}(F)
\quad \text{und} \quad
x \notin F
\]

(sonst gibt es keinen Rundungsfehler).

\vspace{0.5cm}

Dann hat \(x\) die normalisierte \(b\)-adische Darstellung

\[
x = b^{E}\cdot \sum_{i=0}^{\infty} z_i\, b^{-i}.
\]

\vspace{0.3cm}

\textit{(unendliche \(b\)-adische Darstellung)}

\end{frame}

%------------------------------------------------
\begin{frame}{Beweis: Nachbarzahlen in \(F\)}

\scriptsize

Dann sind

\[
x' = b^{E}\sum_{i=0}^{m-1} z_i b^{-i}
\]

und

\[
x'' = b^{E}\left(\sum_{i=0}^{m-1} z_i b^{-i} + b^{-m+1}\right)
\]

in \(F\).

\vspace{0.5cm}

Es gilt:

\[
x' < x < x''
\]

\vspace{0.5cm}

\textbf{Bedeutung:}

\begin{itemize}
\item \(x'\): Abschneiden nach den ersten \(m\) Stellen.
\item \(x''\): dieselben ersten \(m\) Stellen, dann ein Schritt nach oben.
\item Das sind die beiden benachbarten Maschinenzahlen um \(x\).
\end{itemize}

\end{frame}

%------------------------------------------------
\begin{frame}{Beweis: Abschätzung des Rundungsfehlers}

\small

Für jede Rundung \(\operatorname{rd}\) zu \(F\) gilt:

\[
|x-\operatorname{rd}(x)|
\leq
\frac12\,(x''-x')
\]

\vspace{0.4cm}

\[
=
\frac12 \, b^{E}\, b^{-m+1}
\]

\vspace{0.6cm}

\textbf{Warum?}

Der Rundungsfehler ist höchstens halb so groß wie der Abstand  
der beiden benachbarten Maschinenzahlen.

\vspace{0.5cm}

\textbf{Anschaulich:}

\begin{center}
In der Mitte zwischen zwei Maschinenzahlen ist der Fehler am größten.
\end{center}

\end{frame}

%------------------------------------------------
\begin{frame}{Beweis: Untere Schranke für \(x\)}

\scriptsize

Wegen \(z_0>0\) gilt:

\[
x=b^{E}\sum_{i=0}^{\infty} z_i b^{-i}\geq b^{E}.
\]

\vspace{0.5cm}

Denn \(z_0\geq 1\), also ist die Summe mindestens \(1\).

\vspace{0.6cm}

Damit folgt:

\[
x
=
b^{E}+|x-b^{E}|
\geq
b^{E}+|x-\operatorname{rd}(x)|.
\]

\vspace{0.6cm}

Und somit wie gefordert.

\vspace{0.5cm}

\textbf{Idee:}

\begin{itemize}
\item \(x\) liegt rechts von \(b^{E}\).
\item Der Abstand zu \(b^{E}\) ist daher mindestens so groß wie der Rundungsfehler.
\end{itemize}

\end{frame}

%------------------------------------------------
\begin{frame}{Beweis: Abschluss der Abschätzung}

\scriptsize

\[
\left|\frac{x-\operatorname{rd}(x)}{x}\right|
\leq
\frac{|x-\operatorname{rd}(x)|}{b^{E}+|x-\operatorname{rd}(x)|}
\]

\vspace{0.4cm}

\[
=
\frac{1}{1+\dfrac{b^{E}}{|x-\operatorname{rd}(x)|}}
\]

\vspace{0.4cm}

\[
\leq
\frac{1}{1+\dfrac{b^{E}}{\frac12\,b^{E}\,b^{-m+1}}}
\]

\vspace{0.4cm}

\[
=
\frac{1}{1+2b^{m-1}}
\]

\vspace{0.5cm}

\[
\Box
\]

\end{frame}

=
\frac{1}{1+2^{53}}

\]

\vspace{0.6cm}

\end{frame}

\begin{frame}{Hinweis zu \texttt{epsilon()} in C++}

\qquad\text{und}\qquad

\vspace{0.5cm}

Es gilt daher:

\approx

\]

\vspace{0.6cm}

\textbf{Wichtig:}

%------------------------------------------------
\begin{frame}{Definition 4.10: Signifikante Stellen}

\scriptsize

Sei \(F\) ein Maschinenzahlbereich und \(s\in\mathbb{N}\).

Eine Maschinenzahl \(f\in F\) hat \emph{(mindestens) \(s\) signifikante Stellen}
in der \(b\)-adischen Gleitkommadarstellung, falls \(f\neq 0\) und für jede
Rundung \(rd\) sowie jede Zahl \(x\in\mathbb{R}\) mit \(rd(x)=f\) gilt:

\[
|x-f|
\le
\frac{1}{2}\, b^{\lfloor \log_b |f| \rfloor +1-s}
\]

\vspace{0.4cm}

\textbf{Bedeutung:}

\begin{itemize}
\item Der Rundungsfehler darf höchstens so groß sein,
      dass \(s\) Stellen zuverlässig korrekt bleiben.
\item Je größer \(s\), desto genauer ist \(f\).
\item \(\lfloor\cdot\rfloor\) bezeichnet die Gaußklammer
      (Abrunden auf ganze Zahl).
\end{itemize}

\end{frame}

%------------------------------------------------
\begin{frame}{Definition 4.10: Signifikante Stellen}

\scriptsize

Sei \(F\) ein Maschinenzahlbereich und \(s\in\mathbb{N}\).

Eine Maschinenzahl \(f\in F\) hat \emph{(mindestens) \(s\) signifikante Stellen}
in der \(b\)-adischen Gleitkommadarstellung, falls \(f\neq 0\) und für jede
Rundung \(rd\) sowie jede Zahl \(x\in\mathbb{R}\) mit \(rd(x)=f\) gilt:

\[
|x-f|
\le
\frac{1}{2}\, b^{\lfloor \log_b |f| \rfloor +1-s}
\]

\vspace{0.2cm}

\textbf{Wie liest man den Exponenten?}

\[
\lfloor \log_b |f| \rfloor +1-s
\]

\begin{itemize}
\item \(\lfloor \log_b |f| \rfloor\): Größenordnung von \(f\)
      (wo die führende Ziffer liegt)
\item \(+1\): Verschiebung auf die erste signifikante Stelle
\item \(-s\): gehe \(s\) Stellen nach rechts
\item Dadurch trifft man genau die Stelle,
      bis zu der die Zahl sicher korrekt ist.
\end{itemize}

\vspace{0.2cm}

\textbf{Bedeutung:}

Dies ist das mathematische Kriterium dafür, dass die gespeicherte Zahl \(f\)
auf mindestens \(s\) signifikante Stellen genau ist.

\end{frame}

%------------------------------------------------
\begin{frame}{Definition 4.10: Signifikante Stellen mit Beispiel}

\scriptsize

Eine Maschinenzahl \(f\in F\) hat mindestens \(s\) signifikante Stellen, wenn

\[
|x-f|\leq \frac12\, b^{\lfloor \log_b|f|\rfloor+1-s}
\]

gilt.

\vspace{0.2cm}
\textbf{Beispiel:} \(f=1234,\; s=3,\; b=10\)

\[
\lfloor \log_{10}(1234)\rfloor +1-s
=
3+1-3=1
\]

\[
10^1=10
\qquad\Rightarrow\qquad
\frac12\cdot 10=5
\]

Also darf der Fehler höchstens \(5\) sein:

\[
|x-1234|\leq 5
\]

\[
x\in[1229,1239]
\]

\vspace{0.2cm}

\textbf{Interpretation:}

Die ersten \(3\) signifikanten Stellen (\(1,2,3\)) bleiben stabil im
praktischen Sinn.

\vspace{0.2cm}

\textbf{Was sind signifikante Stellen?}

\begin{itemize}
\item \(1234{,}56\) hat \(6\) signifikante Stellen.
\item \(0{,}00452\) hat \(3\) signifikante Stellen (\(4,5,2\)).
\item Führende Nullen zählen nicht.
\end{itemize}

\end{frame}

%------------------------------------------------
\begin{frame}{Beispiel 4.11: Signifikante Stellen bei double}

\scriptsize

Jede Maschinenzahl \(f \in F_{\text{double}} \setminus \{-\max F_{\text{double}},\,0,\,\max F_{\text{double}}\}\)
hat mindestens

\[
\left\lfloor 52 \cdot \log_{10}(2) \right\rfloor = 15
\]

signifikante Stellen in der Dezimaldarstellung.

\vspace{0.3cm}

\textbf{Erklärung:}

\begin{itemize}
\item Double hat \(52\) Bits in der Mantisse (plus hidden bit).
\item Umrechnung Binär \(\rightarrow\) Dezimal:
\[
\log_{10}(2) \approx 0{,}301
\]
\item Daher:
\[
52 \cdot 0{,}301 \approx 15{,}65 \;\Rightarrow\; \lfloor \cdot \rfloor = 15
\]
\end{itemize}

\vspace{0.3cm}

\textbf{Bedeutung:}

Ein double kann etwa \(15\)–\(16\) signifikante Dezimalstellen darstellen.

\end{frame}

 %------------------------------------------------
\begin{frame}{Beispiel 4.11: Fehlerabschätzung bei double}

\scriptsize

Für \(x\in\mathbb{R}\) mit normalisierter Gleitkommadarstellung

\[
x = \sigma \cdot 2^E \cdot \sum_{i=0}^{\infty} z_i\,2^{-i}
\quad \text{und} \quad rd(x)=f
\]

gilt wegen \(2^E \le |f|\):

\[
|x-f|
\le
\frac12 \cdot 2^E \cdot 2^{-52}
=
2^{-53}\,|f|
\]

\vspace{0.3cm}

Umrechnung in Dezimaldarstellung:

\[
2^{-53}
=
10^{-53\log_{10}2}
\approx 10^{-15{,}95}
\]

\[
\Rightarrow\quad
|x-f|
\lesssim
\frac12 \cdot 10^{\lfloor \log_{10}|f|\rfloor +1 - 52\log_{10}2}
\]

\vspace{0.3cm}

\textbf{Schlussfolgerung:}

Der Rundungsfehler eines \texttt{double} ist klein genug,
um etwa \(15\) signifikante Dezimalstellen zu garantieren.

\end{frame}

\begin{frame}{4.4 Maschinenzahlarithmetik}

\small

Sei \(\circ\) eine der vier Operationen

\[
\{+,-,\cdot,/\}.
\]

Wie Beispiel 4.13 verdeutlicht, gilt für \(x,y\in F\) nicht notwendigerweise:

\[
x\circ y \in F.
\]

\vspace{0.5cm}

\textbf{Warum?}

Weil das exakte mathematische Ergebnis oft nicht im endlichen
Maschinenzahlsystem darstellbar ist.

\vspace{0.6cm}

\textbf{Beispiel:}

\[
10^{300}\cdot 10^{300}=10^{600}
\]

Dies liegt außerhalb des erlaubten Exponentenbereichs.

\vspace{0.3cm}

\[
\Rightarrow \text{Overflow / Underflow}
\]

\end{frame}

%------------------------------------------------
\begin{frame}{Ersatzoperation auf dem Rechner}

\small

Stattdessen wird auf dem Rechner eine Ersatzoperation

\[
\odot
\]

ausgeführt, sodass

\[
x\odot y \in F
\]

gilt.

Wir nehmen im Folgenden an, dass:

\vspace{0.6cm}

\textbf{Annahme 4.12}

\[
x\odot y=\operatorname{rd}(x\circ y)
\]

für eine Rundung \(\operatorname{rd}\) zu \(F\) und alle \(x,y\in F\).

\vspace{0.6cm}

\textbf{Bedeutung:}

\begin{itemize}
\item großes Kreiszeichen \(\odot\): Rechneroperation
\item kleines Kreiszeichen \(\circ\): mathematische Operation \(\{+,-,\cdot,/\}\)
\item wichtig, um Computerschritte mathematisch analysieren zu können
\end{itemize}

\end{frame}

\begin{frame}{Hinweis zur Ausarbeitung}

\small

Dies wird zum Beispiel vom IEEE-Standard 754 gefordert.

\vspace{0.5cm}

Zudem wird dies dort auch für die Wurzelfunktion vorausgesetzt,

nicht aber z.\,B. für

\[
\sin(x),\quad \exp(x),\quad \text{etc.}
\]

\vspace{0.8cm}

\textbf{Zum Weiterlesen:}

Die exakte Rundung elementarer Funktionen ist deutlich schwieriger
als bei den vier Grundrechenarten.

\end{frame}

%------------------------------------------------
\begin{frame}{Bedeutung der Rundungsannahme}

\small

Annahme 4.12 garantiert, dass man als Ergebnis einer Rechenoperation
das auf die nächste Maschinenzahl gerundete exakte Ergebnis von \(x\circ y\) erhält.

\vspace{0.4cm}

Man beachte aber, dass zur Berechnung von \(x\odot y\) nicht zuerst
\(x\circ y\) exakt berechnet werden muss.

Vielmehr kann man die Rechnung abbrechen, sobald hinreichend viele Stellen
berechnet sind.

\vspace{0.6cm}

Die Anzahl der Stellen wird durch den Maschinenzahlbereich festgelegt.

\vspace{0.5cm}

\textbf{Beispiel:}

\[
x\circ y = 4{,}613728\ldots
\]

Maschinenzahlen mit 2 Stellen:

\[
4{,}6 \qquad \text{und} \qquad 4{,}7
\]

D.\,h. ab dem Zwischenstand \(4{,}61\) kann man bereits abbrechen
und auf \(4{,}6\) runden.

\vspace{0.3cm}

Sonst rechnet der Computer unnötig lange weiter.

\end{frame}

%------------------------------------------------
\begin{frame}{Beispiel 4.13}

\scriptsize

Sei 
\[
F=F(10,2,-5,5).
\]

Wähle
\[
x=4{,}5\cdot 10^{1}=45
\quad \text{und} \quad
y=1{,}1\cdot 10^{0}=1{,}1.
\]

\vspace{0.4cm}

Dann sind:

\[
x\oplus y=\operatorname{rd}(x+y)
=\operatorname{rd}(46{,}1)
=\operatorname{rd}(4{,}61\cdot 10^{1})
=4{,}6\cdot 10^{1}
\]

\[
x\ominus y=\operatorname{rd}(x-y)
=\operatorname{rd}(43{,}9)
=\operatorname{rd}(4{,}39\cdot 10^{1})
=4{,}4\cdot 10^{1}
\]

\[
x\odot y=\operatorname{rd}(x\cdot y)
=\operatorname{rd}(49{,}5)
=\operatorname{rd}(4{,}95\cdot 10^{1})
\in \{4{,}9\cdot 10^{1},\,5{,}0\cdot 10^{1}\}
\]

\[
x\oslash y=\operatorname{rd}(x/y)
=\operatorname{rd}(40{,}\overline{90})
=\operatorname{rd}(4{,}\overline{09}\cdot 10^{1})
=4{,}1\cdot 10^{1}
\]

\end{frame}

\begin{frame}{Relativer Fehler}

\scriptsize

Gemäß Annahme 4.12 gilt für den relativen Fehler:

\vspace{0.3cm}

\[
\left|
\frac{x\circ y-x\odot y}{x\circ y}
\right|
=
\left|
\frac{x\circ y-\operatorname{rd}(x\circ y)}{x\circ y}
\right|
\leq \operatorname{eps}(F)
\]

\vspace{0.3cm}

falls

\[
|x\circ y|\in \operatorname{range}(F).
\]

\vspace{0.5cm}

Dabei ist

\[
x\odot y=\operatorname{rd}(x\circ y).
\]

\vspace{0.5cm}

Zur Erinnerung:

\[
\operatorname{eps}(F):=
\sup\left\{
\left|
\frac{x-\operatorname{rd}(x)}{x}
\right|
\;\middle|\;
x\in\mathbb{R},\ |x|\in\operatorname{range}(F)
\right\}.
\]

\end{frame}

%------------------------------------------------
\begin{frame}{Beispiel zum relativen Fehler}

\small

\textbf{Beispiel:}

\[
x\circ y = 10{,}00 \quad \text{(exaktes Ergebnis)}
\]

\[
x\odot y = 9{,}99 \quad \text{(gerundetes Ergebnis)}
\]

\vspace{0.5cm}

Absoluter Fehler:
\[
0{,}01
\]

Relativer Fehler:
\[
\frac{0{,}01}{10} = 0{,}001 = 0{,}1\%
\]

\vspace{0.5cm}

Wenn \(\operatorname{eps}(F)=0{,}1\%\), dann passt die Aussage.

\vspace{0.6cm}

\textbf{Merksatz:}

Je kleiner \(\operatorname{eps}(F)\), desto genauer das Zahlensystem.

\end{frame}

%------------------------------------------------
\begin{frame}{Wichtiger Zusatz}

\small

Das Kommutativgesetz für Addition und Multiplikation gilt auch
in der Maschinenzahlarithmetik.

\vspace{0.6cm}

Assoziativ- und Distributivgesetz gelten jedoch
im Allgemeinen nicht.

\vspace{0.8cm}

\[
x\oplus y = y\oplus x
\qquad\qquad
x\odot y = y\odot x
\]

\vspace{0.6cm}

Aber im Allgemeinen nicht:

\[
(x\oplus y)\oplus z = x\oplus (y\oplus z)
\]

\[
x\odot (y\oplus z)= (x\odot y)\oplus (x\odot z)
\]

\end{frame}

%------------------------------------------------
\begin{frame}{Beispiel 4.14}

\scriptsize

Im Maschinenzahlbereich

\[
F=F(10,2,-5,5)
\]

seien

\[
x=4{,}1\cdot 10^{0},\qquad
y=8{,}2\cdot 10^{-1},\qquad
z=1{,}4\cdot 10^{-1}.
\]

\vspace{0.4cm}

Dann sind:

\[
(x\oplus y)\oplus z
=
4{,}9\oplus 0{,}14
=
5{,}0
\]

\[
x\oplus (y\oplus z)
=
4{,}1\oplus 0{,}96
=
5{,}1
\]

\[
x\odot (y\oplus z)
=
4{,}1\odot 0{,}96
=
3{,}9
\]

\[
(x\odot y)\oplus (x\odot z)
=
3{,}4\oplus 0{,}57
=
4{,}0
\]

\vspace{0.4cm}

\textbf{Warum ist das so?}

Nach jedem Rechenschritt wird gerundet:

\[
\text{Rechnen} \;\rightarrow\; \text{Runden} \;\rightarrow\; \text{Weiterrechnen}
\]

Dadurch verändern sich Zwischenwerte.

\end{frame}

%------------------------------------------------
\begin{frame}{Wichtige Erkenntnis}

\small

Folglich können mathematisch äquivalente Ausdrücke bei der Ausführung
in der Maschinenzahlarithmetik zu wesentlich unterschiedlichen Ergebnissen führen,
auch wenn der Input als Maschinenzahlen darstellbar ist.

\vspace{0.4cm}

Dies kann \textbf{dramatische Folgen} haben.

\vspace{0.6cm}

\textbf{Übersetzung:}

Auch wenn zwei Formeln mathematisch exakt dasselbe bedeuten,
können sie auf dem Computer verschiedene Ergebnisse liefern.

\vspace{0.6cm}

\textbf{Warum? Wegen:}

\begin{itemize}
\item Rundung
\item Reihenfolge der Rechenschritte
\item Auslöschungseffekten
\item Overflow / Underflow
\item Verlust von Stellen
\end{itemize}

\end{frame}

%------------------------------------------------
\begin{frame}{Folgen für die Praxis}

\small

\textbf{Folgen für die Praxis:}

\begin{itemize}
\item falsche Simulationen
\item falsche wissenschaftliche Ergebnisse
\item Abstürze
\item ungenaue Finanzberechnungen
\item Fehler in Technik / Navigation / Medizinsoftware
\end{itemize}

\vspace{0.8cm}

\textbf{Wichtigste Erkenntnis des Kapitels:}

In numerischer Mathematik zählt nicht nur,
ob eine Formel mathematisch richtig ist,

sondern auch, \textbf{wie} man sie berechnet.

\end{frame}

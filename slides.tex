%------------------------------------------------
\begin{frame}{4.1 Grundidee}
Ziel: Jede reelle Zahl soll in einer Basis $b\geq 2$ dargestellt werden.

\bigskip

Bekannte Beispiele:
\begin{itemize}
\item $b=10$ : Dezimalsystem
\item $b=2$ : Binärsystem
\item $b=16$ : Hexadezimalsystem
\end{itemize}

\bigskip

Wir suchen also eine Darstellung, die für jede Zahl funktioniert.
\end{frame}

%------------------------------------------------
\begin{frame}{Satz zur Darstellung}
\small
\begin{thm}
Sei $b\in\N$, $b\geq 2$, und $x\in\R\setminus\{0\}$.

Dann gibt es eindeutig bestimmte Zahlen
\[
E\in\Z,\qquad \sigma\in\{-1,1\},\qquad z_i\in\{0,1,\dots,b-1\},
\]
so dass
\[
x=\sigma\cdot b^E\cdot\left(\sum_{i=0}^{\infty} z_i b^{-i}\right).
\]
\end{thm}

Diese Darstellung heißt:

\bigskip

\textbf{normalisierte b-adische Darstellung von $x$}
\end{frame}

%------------------------------------------------
\begin{frame}{Die Formel einfach erklärt}
\[
x=\sigma\cdot b^E\cdot\left(\sum_{i=0}^{\infty} z_i b^{-i}\right)
\]

\bigskip

\begin{itemize}
\item $\sigma$ = Vorzeichen der Zahl
\item $b$ = Basis des Zahlensystems
\item $E$ = Exponent / Größenordnung
\item $z_i$ = Ziffern der Zahl
\item $b^{-i}$ = Stellenwert
\end{itemize}

\bigskip

Also: Vorzeichen $\times$ Größe $\times$ eigentliche Zifferndarstellung
\end{frame}

%------------------------------------------------
\begin{frame}{Bedeutung der Ziffern}
Die Ziffern erfüllen:
\[
z_i\in\{0,1,\dots,b-1\}
\]

Das heißt:

\begin{itemize}
\item Im Dezimalsystem: $0$ bis $9$
\item Im Binärsystem: nur $0$ oder $1$
\item Im Hexsystem: $0$ bis $15$
\end{itemize}

\bigskip

Jede Ziffer wird mit ihrem Stellenwert multipliziert:
\[
z_i\cdot b^{-i}
\]

Dann werden alle Beiträge addiert.
\end{frame}

%------------------------------------------------
\begin{frame}{Warum die unendliche Summe?}
Wir schreiben:
\[
\sum_{i=0}^{\infty} z_i b^{-i}
\]

Das bedeutet eine unendliche Reihe.

\bigskip

Formal:
\[
\sum_{i=0}^{\infty} a_i
=
\lim_{n\to\infty}\sum_{i=0}^{n} a_i
\]

\bigskip

Man betrachtet also immer längere Teilsummen.
\end{frame}

%------------------------------------------------
\begin{frame}{Warum existiert der Grenzwert?}
\begin{itemize}
\item Alle Summanden sind $\geq 0$
\item Die Teilsummen wachsen also nur nach oben
\item Gleichzeitig sind sie beschränkt
\end{itemize}

Denn:
\[
z_i\leq b-1
\]

also:
\[
\sum_{i=0}^{\infty} z_i b^{-i}
\leq
\sum_{i=0}^{\infty}(b-1)b^{-i}
\]

Dadurch existiert der Grenzwert.
\end{frame}

%------------------------------------------------
\begin{frame}{Wichtige Abschätzung}
Es gilt:
\[
1\leq \sum_{i=0}^{\infty} z_i b^{-i}<b
\]

\bigskip

Warum?

\begin{itemize}
\item Untere Grenze: $z_0\neq 0$
\item Obere Grenze: keine Ziffer ist größer als $b-1$
\end{itemize}

\bigskip

Die Mantisse liegt also immer zwischen $1$ und $b$.
\end{frame}

%------------------------------------------------
\begin{frame}{Beginn des Beweises: Vorzeichen}
Das Vorzeichen ist direkt klar:

\[
\sigma=
\begin{cases}
-1,& x<0\\[0.2cm]
1,& x>0
\end{cases}
\]

\bigskip

Danach betrachtet man nur noch den Betrag $|x|$.
\end{frame}

%------------------------------------------------
\begin{frame}{Normierung der Zahl}
Wir wollen die Zahl in den Bereich $[1,b)$ bringen.

Dazu wählen wir:
\[
E:=\lfloor \log_b |x|\rfloor
\]

und setzen:
\[
a_0:=b^{-E}|x|
\]

Dann gilt:
\[
1\leq a_0<b
\]

\bigskip

$a_0$ nennt man die normierte Mantisse.
\end{frame}

%------------------------------------------------
\begin{frame}{Warum macht man das?}
Dann kann man schreiben:
\[
x=\sigma\cdot b^E\cdot a_0
\]

\bigskip

Das trennt die Zahl in zwei Teile:

\begin{itemize}
\item $b^E$ = Größe der Zahl
\item $a_0$ = eigentliche Ziffernstruktur
\end{itemize}

Das ist genau dieselbe Idee wie wissenschaftliche Schreibweise.
\end{frame}

%------------------------------------------------
\begin{frame}{Ziffern Schritt für Schritt bestimmen}
Nun gewinnen wir die Ziffern aus $a_0$.

Für $i\in\N\cup\{0\}$:
\[
z_i=\lfloor a_i\rfloor
\]

\[
a_{i+1}=b(a_i-z_i)
\]

\bigskip

Bedeutung:

\begin{itemize}
\item Ganzzahliger Teil = nächste Ziffer
\item Rest = Nachkommateil
\item Mit $b$ multiplizieren = nächste Stelle nach vorne holen
\end{itemize}
\end{frame}

%------------------------------------------------
\begin{frame}{Beispiel in Basis 10}
Sei:
\[
a_0=3{,}14
\]

Dann:

\[
z_0=\lfloor 3{,}14\rfloor=3
\]

\[
a_1=10(3{,}14-3)=1{,}4
\]

\[
z_1=\lfloor 1{,}4\rfloor=1
\]

\[
a_2=10(1{,}4-1)=4
\]

Also entstehen die Ziffern $3,1,4,\dots$
\end{frame}

%------------------------------------------------
\begin{frame}{Fazit zu 4.1}
\begin{itemize}
\item Jede reelle Zahl besitzt eine b-adische Darstellung
\item Die Basis kann frei gewählt werden
\item Durch Normierung erhält man eine einheitliche Form
\item Die Ziffern werden rekursiv berechnet
\end{itemize}

\bigskip

Wichtig für später:

\bigskip

Darauf bauen Maschinenzahlen und Computerarithmetik auf.
\end{frame}

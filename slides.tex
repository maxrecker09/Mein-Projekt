    \item Jede reelle Zahl soll in einer Basis \( b \geq 2 \) dargestellt werden.
\end{itemize}

\[
\begin{array}{l}
b \in \N,\; b \geq 2 \\[0.3em]

b = 2:\ \text{Binärsystem} \\

b = 16:\ \text{Hexadezimalsystem}
\end{array}
\]

\vspace{0.5cm}

\begin{frame}{Satz 4.1: b-adische Darstellung reeller Zahlen}

\begin{thm}
Sei \( b \in \N \) mit \( b \geq 2 \). Sei \( x \in \R \setminus \{0\} \). Dann existieren eindeutig bestimmte Zahlen

\[
E \in \Z,\quad \sigma \in \{-1,1\},\quad z_i \in \{0,1,\dots,b-1\}
\]

für \( i \in \N \cup \{0\} \) mit

\[
\{\, i \in \N \mid z_i \neq b-1 \,\} \text{ unendlich}, \quad z_0 \neq 0
\]

und

\[
x=\sigma \cdot b^E \cdot \left(\sum_{i=0}^{\infty} z_i \, b^{-i}\right).
\]

Dies heißt die (normalisierte) b-adische Darstellung von \(x\).
\end{thm}

\end{frame}

%------------------------------------------------
\begin{frame}{Bestandteile von \(x\)}

\[
x=\sigma \cdot b^E \cdot \left(\sum_{i=0}^{\infty} z_i\, b^{-i}\right)
\]

\vspace{0.4cm}

\begin{itemize}
    \item \(x\): die Zahl, die dargestellt werden soll
    
    \item \(\sigma \in \{-1,1\}\): Vorzeichen
    
    \item \(b\): Basis des Zahlensystems
    
    \item \(E \in \Z\): Exponent / Größenordnung
    
    \item \(z_i\): \(i\)-te Ziffer der Zahl
    
    \item \(b^{-i}\): Stellenwert der jeweiligen Ziffer
\end{itemize}

\vspace{0.4cm}

\textbf{Beispiel:}

\[
3{,}1415
\quad\Rightarrow\quad
z_0=3,\; z_1=1,\; z_2=4,\; z_3=1,\; z_4=5
\]

\vspace{0.3cm}

\[
\Rightarrow\quad
\text{Zahl = Vorzeichen \(\cdot\) Größe \(\cdot\) Mantisse}
\]

\end{frame}

\begin{frame}{Behauptung und Beginn des Beweises}

\tiny

\textbf{Behauptung:} Es gilt:

\[
1 \leq \sum_{i=0}^{\infty} z_i b^{-i}
<
\sum_{i=0}^{\infty} (b-1)b^{-i}
= b,
\]

d.h. der Grenzwert existiert,

weil die Folge monoton wachsend ist und durch \(b\) nach oben beschränkt.

\vspace{0.4cm}

\textbf{Bew. der Aussage:}

\begin{enumerate}
\item Sei \(x\in\R\setminus\{0\}\).

Offenbar muss \(\sigma=-1\) gelten falls \(x<0\) und
\(\sigma=1\) falls \(x>0\).

Für

\[
z_i\in\{0,1,\dots,b-1\}
\qquad (i\in\N\cup\{0\})
\]

mit

\[
\{\,i\in\N \mid z_i\neq b-1\,\}\neq\emptyset
\qquad \text{und} \qquad z_0\neq 0
\]

gilt.
\end{enumerate}

\end{frame}

\begin{frame}{Warum gilt \(1 \leq \sum z_i b^{-i}\)?}

\tiny

\textbf{(2) Warum ist}
\[
1 \leq \sum_{i=0}^{\infty} z_i b^{-i}\; ?
\]

Da \(z_0 \neq 0 \Rightarrow z_0 \geq 1\), da

\[
z_i \in \{0,1,\dots,b-1\}
\]

\[
\Rightarrow \sum_{i=0}^{\infty} z_i b^{-i} \geq 1
\]

und nur falls gilt:

\[
\forall i\in\N:\ z_i=0
\]

\[
\Rightarrow \sum_{i=0}^{\infty} z_i b^{-i} = 1,
\]

sonst immer größer als \(1\).

\end{frame}

%------------------------------------------------
\begin{frame}{Warum ist \(\sum (b-1)b^{-i}=b\)?}

\tiny

\textbf{(3) Warum ist}
\[
\sum_{i=0}^{\infty} (b-1)b^{-i} = b\; ?
\]

\begin{itemize}
\item \((b-1)\): letztes Glied von \(z_i\)
\end{itemize}

\[
= (b-1)\cdot \sum_{i=0}^{\infty} b^{-i}
\]

\[
= (b-1)\cdot \frac{1}{1-\frac{1}{b}}
= (b-1)\cdot \left(1-\frac{1}{b}\right)^{-1}
\]

\[
= (b-1)\cdot \frac{b}{b-1} = b
\]

\end{frame}

\begin{frame}{Ziel der Normierung}

\textbf{Behauptung:}
\[
E := \left\lfloor \log_b |x| \right\rfloor
\]

\vspace{0.4cm}

Wir wollen \(x\) so schreiben, dass

\[
x=\sigma \cdot b^E \cdot a_0
\qquad \text{mit} \qquad 1 \leq a_0 < b.
\]

Dann folgt:

\[
a_0=b^{-E}|x|.
\]

\vspace{0.4cm}

Gesucht ist also ein Exponent \(E\), für den gilt:

\[
1 \leq b^{-E}|x| < b.
\]

Umformen liefert:

\[
1 \leq \frac{|x|}{b^E} < b
\quad \Longleftrightarrow \quad
b^E \leq |x| < b^{E+1}.
\]

\vspace{0.4cm}

Das bedeutet:

\begin{itemize}
    \item Finde eine ganze Zahl \(E\), sodass \(|x|\)
    zwischen zwei benachbarten Potenzen von \(b\) liegt.
\end{itemize}

\end{frame}

%------------------------------------------------
\begin{frame}{Bestimmung des Exponenten \(E\)}

Aus

\[
b^E \leq |x| < b^{E+1}
\]

wenden wir den Logarithmus zur Basis \(b\) an:

\[
\log_b(b^E) \leq \log_b(|x|) < \log_b(b^{E+1}).
\]

Mit

\[
\log_b(b^E)=E,
\qquad
\log_b(b^{E+1})=E+1
\]

folgt:

\[
E \leq \log_b(|x|) < E+1.
\]

\vspace{0.4cm}

Damit muss gelten:

\[
E=\left\lfloor \log_b(|x|)\right\rfloor.
\]

\vspace{0.5cm}

\textbf{Grund:}

Für

\[
a_0=b^{-E}|x|=\frac{|x|}{b^E}
\]

gilt dann automatisch:

\[
1 \leq a_0 < b.
\]

Also liefert
\[
E=\left\lfloor \log_b(|x|)\right\rfloor
\]
genau den passenden Exponenten für die normierte Darstellung.

\end{frame}

\begin{frame}{Darstellung von \(a_0\) mit Restglied}

\scriptsize

\textbf{Behauptung:} Für alle \(n\in\N\cup\{0\}\) gilt

\[
a_0=
\sum_{i=0}^{n} z_i b^{-i}
+
a_{n+1}b^{-n-1}.
\]

\vspace{0.2cm}

Dabei ist:

\begin{itemize}
\item \(\sum_{i=0}^{n} z_i b^{-i}\): bekannte Ziffern / Partialsumme
\item \(a_{n+1}b^{-n-1}\): Restglied
\end{itemize}

\vspace{0.2cm}

Der verbleibende Rest wird ab dem Index \(n+1\) zusammengefasst.

\vspace{0.3cm}

\textbf{Ziel:}

\[
a_0=\sum_{i=0}^{\infty} z_i b^{-i}.
\]

Das Restglied verschwindet für \(n\to\infty\).

\vspace{0.3cm}

\textbf{Wichtig:}

Falls

\[
1\leq x<b,
\]

dann gilt bereits

\[
a_0=x,
\]

also ist keine zusätzliche Normierung nötig.

\end{frame}

\begin{frame}{Beweis per Induktion: Induktionsanfang}

\small

\textbf{Induktionsanfang: \(n=0\)}

Zu zeigen:

\[
a_0=z_0 b^{0}+a_1 b^{-1}.
\]

\[
a_0=z_0+\frac{a_1}{b}.
\]

Wir wissen:

\[
a_1=b(a_0-z_0).
\]

Einsetzen liefert:

\[
a_0=
z_0+\frac{b(a_0-z_0)}{b}.
\]

\[
a_0=
z_0+(a_0-z_0)=a_0.
\]

\hfill \(\checkmark\)

\end{frame}

%------------------------------------------------
\begin{frame}{Beweis per Induktion: Induktionsvoraussetzung}

\small

\textbf{Induktionsvoraussetzung \((n-1\rightarrow n)\):}

Zu zeigen:

\[
a_0=
\sum_{i=0}^{n} z_i b^{-i}
+
a_{n+1} b^{-n-1}.
\]

\vspace{0.5cm}

Also lautet die Induktionsannahme:

\[
a_0=
\sum_{i=0}^{n-1} z_i b^{-i}
+
a_n b^{-n}.
\]

\end{frame}

%------------------------------------------------
\begin{frame}{Beweis per Induktion: Induktionsschritt}

\small

\textbf{Induktionsschritt (Rekursionsformel):}

Aus

\[
a_{n+1}=b(a_n-z_n)
\]

folgt:

\[
a_n=z_n+\frac{a_{n+1}}{b}.
\]

\vspace{0.4cm}

Einsetzen in die Induktionsannahme:

\[
a_0=
\sum_{i=0}^{n-1} z_i b^{-i}
+
\left(z_n+\frac{a_{n+1}}{b}\right)b^{-n}.
\]

\vspace{0.4cm}

Umformen ergibt:

\[
a_0=
\sum_{i=0}^{n-1} z_i b^{-i}
+
z_n b^{-n}
+
a_{n+1} b^{-n-1}.
\]

\vspace{0.4cm}

Also:

\[
a_0=
\sum_{i=0}^{n} z_i b^{-i}
+
a_{n+1} b^{-n-1}.
\]

\hfill \(\checkmark\)

\end{frame}

\begin{frame}{Übergang zum Grenzwert}

\small

Aus

\[
x=\sigma \cdot b^E \cdot a_0
\]

und

\[
a_0=
\sum_{i=0}^{n} z_i b^{-i}
+
a_{n+1}b^{-n-1}
\]

für jedes \(n\in\N\cup\{0\}\) folgt:

\[
x=
\sigma \cdot b^E \cdot
\lim_{n\to\infty}
\left(
\sum_{i=0}^{n} z_i b^{-i}
\right),
\]

sofern das Restglied verschwindet.

\vspace{0.3cm}

Also müssen wir zeigen:

\[
a_{n+1}b^{-n-1}\to 0
\qquad (n\to\infty).
\]

\vspace{0.3cm}

Per Induktion gilt die Darstellung bereits für jedes endliche \(n\).  
Nun geht man zum Grenzwert über.

\end{frame}

%------------------------------------------------
\begin{frame}{Verschwinden des Restglieds}

\small

Zu zeigen:

\[
a_{n+1}b^{-n-1}\to 0.
\]

Dann bleibt:

\[
a_0=
\lim_{n\to\infty}\sum_{i=0}^{n} z_i b^{-i}
=
\sum_{i=0}^{\infty} z_i b^{-i}.
\]

\vspace{0.4cm}

Es wurde gezeigt:

\[
1\leq a_{n+1}<b.
\]

Daraus folgt:

\[
0\leq a_{n+1}b^{-n-1}
<
b\cdot b^{-n-1}
=
b^{-n}.
\]

Da \(b^{-n}\to 0\), gilt mit dem Sandwichprinzip:

\[
a_{n+1}b^{-n-1}\to 0.
\]

\end{frame}

\begin{frame}{Eindeutigkeit der normalisierten Darstellung}

\small

\textbf{Annahme:}

\[
\{\, i\in\N \mid z_i\neq b-1 \,\}
\]

sei endlich.

Dann existiert ein \(n_0\in\N\) mit

\[
z_i=b-1
\qquad \text{für alle } i>n_0.
\]

\vspace{0.4cm}

\textbf{Behauptung:}

Für jedes \(x\in\R\setminus\{0\}\) gibt es genau eine normalisierte Darstellung

\[
x=\sigma \cdot b^E \cdot \sum_{i=0}^{\infty} z_i b^{-i},
\]

mit

\[
\sigma\in\{-1,1\},\quad E\in\Z,\quad z_0\neq 0.
\]

Die Bedingung verhindert doppelte Darstellungen mit endlosem \(b-1\)-Schwanz.

\end{frame}

%------------------------------------------------
\begin{frame}{1. Existenz der Darstellung}

\textbf{Existenz:}

Zu jeder Zahl \(x\) lassen sich passende Ziffern \(z_i\) wählen, sodass

\[
x=\sigma \cdot b^E \cdot \sum_{i=0}^{\infty} z_i b^{-i}
\]

gilt.

\vspace{0.5cm}

\textbf{Idee des Beweises:}

\begin{itemize}
    \item Die Zahl wird schrittweise in Basis \(b\) zerlegt.
    
    \item Für jede Stelle wird eine passende Ziffer \(z_i\) bestimmt.
    
    \item Dadurch entsteht die unendliche Reihe
    
    \[
    \sum_{i=0}^{\infty} z_i b^{-i}.
    \]
    
    \item Würde ab einem gewissen Punkt nur noch \(b-1\) auftreten, entstünde eine alternative Darstellung.
    
    \item Deshalb wird dieser Fall ausgeschlossen.
\end{itemize}

\vspace{0.5cm}

\[
\Rightarrow \text{Also existiert eine normalisierte Darstellung.}
\]

\end{frame}

%------------------------------------------------
\begin{frame}{2. Eindeutigkeit der Darstellung}

\textbf{Beweisidee zur Eindeutigkeit:}

Angenommen, es gäbe zwei verschiedene normalisierte Darstellungen derselben Zahl.

\vspace{0.4cm}

Dann sei \(n\) die erste Stelle, an der sich die Ziffern unterscheiden.

\begin{itemize}
    \item Vor der Stelle \(n\) sind beide Darstellungen identisch.
    
    \item An der Stelle \(n\) ist eine Darstellung größer als die andere.
    
    \item Dieser Unterschied besitzt den Stellenwert \(b^{-n}\).
    
    \item Die späteren Stellen haben nur kleinere Gewichte
    \(\left(b^{-n-1}, b^{-n-2}, \dots \right)\)
    und können den Unterschied nicht mehr vollständig ausgleichen.
\end{itemize}

\vspace{0.5cm}

Daher können beide Summen nicht gleich sein.

\[
\Rightarrow \text{Die Darstellung ist eindeutig.}
\]

\end{frame}

%------------------------------------------------
\begin{frame}{Merksatz und Beispiele}

\textbf{Merksatz:}

Wenn an der ersten unterschiedlichen Stelle eine Ziffer kleiner ist, dann kann der Rest diesen Unterschied nie vollständig aufholen.

\vspace{0.6cm}

\textbf{Beispiele für \(b=2\):}

\[
\frac{1}{3}
=
2^{-2}\left(2^0+2^{-2}+2^{-4}+2^{-6}+\dots\right)
=
(0,\overline{01})_2
\]

\vspace{0.6cm}

\[
8{,}1
=
2^3\left(2^0+2^{-7}+2^{-8}+2^{-11}+2^{-12}+\dots\right)
=
(1000,0\overline{0011})_2
\]

\end{frame}
